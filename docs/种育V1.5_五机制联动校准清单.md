## 校准项 1：换球心反例保护——时间尺度参数化

### 问题
纯粹度 = 保持×时间×抗摇摆。技能尺度培育第 3 周时间项 ≈0.5，纯粹度必然 <0.7 → 必然 WARN → 连续 2 轮后误触发换球心。反例保护"持续 >5 年"是人才尺度，技能尺度下失效。

### 修改点

```python
# 反例保护按时间尺度缩放（原：>5 年固定值）
def _thickening_period_protection(duration, time_scale):
    """
    厚积期反例保护：按时间尺度缩放
    人才尺度（10-25年）：持续 >5 年 且 抗摇摆 → 判定厚积期
    技能尺度（1-6月）：持续 >2 个月 且 抗摇摆 → 判定积累期
    """
    thresholds = {
        "talent": 5.0,   # 年
        "skill": 2.0,    # 月
    }
    return duration >= thresholds[time_scale]
```

### 测试用例

```python
def test_skill_scale_protection():
    """技能尺度：持续 2.5 个月 + 抗摇摆 → 不触发换球心（积累期保护）"""
    seed = Seed(duration_months=2.5, anti_sway=1.0, time_scale="skill")
    assert _thickening_period_protection(seed.duration, seed.time_scale) is True

def test_skill_scale_no_protection():
    """技能尺度：持续 1.5 个月（<2月）→ 保护不生效，纯粹度 <0.7 连续 2 轮 → 触发"""
    seed = Seed(duration_months=1.5, time_scale="skill")
    assert _thickening_period_protection(seed.duration, seed.time_scale) is False

def test_talent_scale_unchanged():
    """人才尺度：4 年（<5年）→ 不保护；6 年 → 保护"""
    assert _thickening_period_protection(4.0, "talent") is False
    assert _thickening_period_protection(6.0, "talent") is True
```

## 校准项 2：饱和检测——跳过留白轮

### 问题
留白轮只观察不产出，边际产出天然为 0 → 判"已饱和"是留白的正常结果，不是日益饱和。留白与饱和同时亮灯，信号矛盾。

### 修改点

```python
def _marginal_output_check(rounds, current_round):
    """
    饱和检测：跳过留白轮
    留白轮的零产出不计入边际产出序列
    """
    productive_rounds = [r for r in rounds if r.type != "leave_blank"]  # 排除留白轮
    if len(productive_rounds) < 3:
        return {"saturated": False, "reason": "有效产出轮不足 3 轮"}
    recent = productive_rounds[-3:]
    marginal = (recent[-1].output - recent[0].output) / 3
    return {"saturated": marginal < 0.1, "marginal": marginal}
```

### 测试用例

```python
def test_leave_blank_round_not_counted():
    """留白轮零产出不计入：留白轮后边际产出不受污染"""
    rounds = [Round(type="productive", output=1.0), Round(type="productive", output=1.05),
              Round(type="leave_blank", output=0.0), Round(type="productive", output=1.10)]
    result = _marginal_output_check(rounds, current_round=4)
    assert result["saturated"] is False  # 有效轮边际 (1.10-1.0)/3 = 0.033 < 0.1 但仅 3 轮有效 → 不足 4 轮？按规则检查
    # 注：有效轮 = 3 个（1.0/1.05/1.10），边际 = 0.033 < 0.1 → 触发饱和；若把留白轮计入则会误判为 0 边际
    # 测试重点是：留白轮 0.0 不进入 productive_rounds

def test_saturation_triggers_after_3_productive():
    """连续 3 轮有效产出边际 <0.1 → 触发转日损"""
    rounds = [Round(type="productive", output=1.0), Round(type="productive", output=1.01),
              Round(type="productive", output=1.02)]
    result = _marginal_output_check(rounds, current_round=3)
    assert result["saturated"] is True
```

## 校准项 3：失败质量审核——三分法（真失败/偶然失败/表演性失败）

### 问题
当前"可复现=False"→ 判"表演性失败"。但不可复现是"偶然失败"（运气/环境），表演性失败是"为奖励故意制造"——动机完全不同，语义混淆。

### 修改点

```python
class FailureQuality(str, Enum):
    TRUE = "真失败"        # 结构性证伪：证伪条件=True + 可复现=True + 修正方向=True → 奖励（入证伪库）
    ACCIDENTAL = "偶然失败"  # 不可复现（运气/环境）→ 中性，不奖惩
    PERFORMATIVE = "表演性失败"  # 故意为之（为奖励制造）→ 警告

def _classify_failure(falsified, reproducible, correction_dir, intent_hint):
    if falsified and reproducible and correction_dir:
        return FailureQuality.TRUE
    if not reproducible:
        return FailureQuality.ACCIDENTAL  # 不可复现 → 偶然（原版误判为表演性）
    if intent_hint == "deliberate":
        return FailureQuality.PERFORMATIVE
    return FailureQuality.ACCIDENTAL
```

### 测试用例

```python
def test_true_failure_rewarded():
    """结构性证伪（可复现）→ 真失败 → 奖励入证伪库"""
    q = _classify_failure(falsified=True, reproducible=True, correction_dir=True, intent_hint=None)
    assert q == FailureQuality.TRUE

def test_unreproducible_is_accidental():
    """不可复现 → 偶然失败（非表演性）→ 中性不奖惩"""
    q = _classify_failure(falsified=True, reproducible=False, correction_dir=True, intent_hint=None)
    assert q == FailureQuality.ACCIDENTAL  # 修正：原版判表演性

def test_deliberate_is_performative():
    """故意为之 → 表演性失败 → 警告"""
    q = _classify_failure(falsified=False, reproducible=True, correction_dir=False, intent_hint="deliberate")
    assert q == FailureQuality.PERFORMATIVE
```

## 校准项 4：减法统计口径——声明构成

### 问题
"全程累计 13 条"→"18 条"未声明构成；9.4 表"培育级 6 + 协议级 5 = 11"与 18 不符。

### 修改点

```python
def _subtraction_summary(rounds, protocol_subtractions):
    """
    减法统计口径统一：
    全程累计 = 培育级累计（跨轮次）+ 协议级（V1.4→V1.5 减除）
    """
    cultivation_total = sum(r.subtractions for r in rounds)  # 培育级全程累计（13）
    protocol_total = len(protocol_subtractions)              # 协议级（5）
    return {
        "cultivation_total": cultivation_total,
        "protocol_total": protocol_total,
        "grand_total": cultivation_total + protocol_total,   # 18
        "note": f"全程累计 = 培育级 {cultivation_total} + 协议级 {protocol_total} = {cultivation_total + protocol_total}",
    }
```

### 测试用例

```python
def test_subtraction_breakdown():
    """全程累计构成明确：培育级 13 + 协议级 5 = 18"""
    s = _subtraction_summary(rounds=[Round(subtractions=13)], protocol_subtractions=[1]*5)
    assert s["cultivation_total"] == 13
    assert s["protocol_total"] == 5
    assert s["grand_total"] == 18
    assert "培育级 13 + 协议级 5 = 18" in s["note"]
```

## 校准项 5：审计口径统一——并列清单改用纯粹度

### 问题
M3 并列清单 B 案例用旧口径"七维骨架完整≡通过"，§5.2 用新口径纯粹度（0.500 WARN）——同一报告两套性决定审计并存。

### 修改点

```python
def _promotion_list(cases, purity_results):
    """
    并列清单统一到纯粹度口径：
    晋升判定 = 宪法审计 PASS + 纯粹度 ≥0.7（或"积累期保护"标注）
    """
    for case in cases:
        purity = purity_results[case.id]
        protected = _thickening_period_protection(case.duration, case.time_scale)
        if case.constitution_audit != "PASS":
            case.promotion = "❌ 宪法审计未过"
        elif purity >= 0.7 or protected:
            case.promotion = "✅ 晋升（纯粹度 or 积累期保护）"
        else:
            case.promotion = "⏳ 待观察（纯粹度 <0.7，未达保护）"
        case.audit_method = "purity_v1.5"  # 统一标注审计口径
```

### 测试用例

```python
def test_promotion_unified_to_purity():
    """并列清单统一用纯粹度：B 案例（0.500 + 积累期保护）→ 晋升"""
    case_b1 = Case(id="B-1", constitution_audit="PASS", duration_months=2.5, time_scale="skill")
    result = _promotion_list([case_b1], {"B-1": 0.500})
    assert result[0].audit_method == "purity_v1.5"   # 口径统一
    assert result[0].promotion == "✅ 晋升（纯粹度 or 积累期保护）"

def test_promotion_waits_without_protection():
    """无积累期保护 + 纯粹度 <0.7 → 待观察（不晋升）"""
    case = Case(id="X", constitution_audit="PASS", duration_months=1.5, time_scale="skill")
    result = _promotion_list([case], {"X": 0.500})
    assert "待观察" in result[0].promotion
```

## 校准项 6（P3 顺手清理）：S_p 重复 + D6 同步

### 修改点

```python
# S_p 判语只保留一个精度（保留精确值，删除四舍五入重复）
def _format_verdict(phase, sp_value):
    return f"[低信度] [萌芽前] {phase}倾向（S_p={sp_value:.2f}，未入阶段）"  # 只输出一次

# D6 与 D5 同步：并列主导时画像提示也用并列清单
def _portrait_hint(dominant, n, parallel_list=None):
    if parallel_list:
        return f"画像提示（{'/'.join(parallel_list)}并列主导，n={n}）"  # 与 D5 同步
    return f"画像提示（{dominant}主导，n={n}）"
```

### 测试用例

```python
def test_verdict_single_sp():
    """判语只含一个 S_p 值"""
    v = _format_verdict("火·化", 12.93)
    assert v.count("12.93") == 1 and "12.9" not in v

def test_portrait_sync_with_parallel():
    """D6 与 D5 并列同步"""
    p = _portrait_hint("火", 3, parallel_list=["火", "土", "水"])
    assert "火/土/水并列主导" in p
```
